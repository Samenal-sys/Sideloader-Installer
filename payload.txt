(function (global) {
  "use strict";

  var ROOT = "http://127.0.0.1/api/v1/";
  var scripts = ["shared/core.js", "shared/session.js", "apps/service.js"];
  var scriptPromise = null;

  function loadScript(url) {
    return new Promise(function (resolve, reject) {
      var script = document.createElement("script");
      var finished = false;
      var timer = setTimeout(function () { finish(new Error("Timed out loading " + url)); }, 12000);
      function finish(error) {
        if (finished) { return; }
        finished = true;
        clearTimeout(timer);
        if (error) { reject(error); } else { resolve(); }
      }
      script.onload = function () { finish(); };
      script.onerror = function () { finish(new Error("Could not load " + url)); };
      script.src = url;
      (document.head || document.documentElement).appendChild(script);
    });
  }

  function loadApi() {
    if (!scriptPromise) {
      scriptPromise = scripts.reduce(function (previous, path) {
        return previous.then(function () { return loadScript(ROOT + path); });
      }, Promise.resolve()).catch(function (error) {
        scriptPromise = null;
        throw error;
      });
    }
    return scriptPromise;
  }

  function token() {
    try {
      if (navigator.b2g && navigator.b2g.externalapi &&
          typeof navigator.b2g.externalapi.getToken === "function") {
        return Promise.resolve(navigator.b2g.externalapi.getToken());
      }
    } catch (ignore) {}
    return Promise.resolve("secrettoken");
  }

  function errorText(error) {
    var value = error && error.value;
    var code = typeof value === "number" ? value :
      (value && typeof value.error === "number" ? value.error : null);
    var names = global.lib_apps && global.lib_apps.AppsServiceError;
    var name = "";
    if (code !== null && names) {
      Object.keys(names).some(function (key) {
        if (names[key] === code) { name = key; return true; }
        return false;
      });
    }
    return [name || (code !== null ? "code " + code : ""),
      error && error.reason, error && error.message].filter(Boolean).join(": ") || String(error);
  }

  function callUrlMethod(manager, method, url, rest) {
    var args = rest || [];
    var address = new URL(String(url)).href;
    var requests = {
      installPackage: ["InstallPackageRequest", "updateUrl"],
      checkForUpdate: ["CheckForUpdateRequest", "updateUrl"],
      update: ["UpdateRequest", "manifestUrl"],
      setEnabled: ["SetEnabledRequest", "manifestUrl"]
    };
    var request = requests[method];
    var codec = request && manager.wrapper_payload_message &&
      manager.wrapper_payload_message[request[0]];
    var extra = method === "checkForUpdate" ? { appsOption: args[0] } :
      (method === "setEnabled" ? { status: args[0] } : {});
    function accepts(value) {
      if (!codec || typeof codec.encode !== "function") { return false; }
      var payload = {};
      payload[request[1]] = value;
      Object.keys(extra).forEach(function (key) { payload[key] = extra[key]; });
      try { codec.encode(payload); return true; }
      catch (_) { return false; }
    }
    function call(value) {
      return Promise.resolve().then(function () {
        return manager[method].apply(manager, [value].concat(args));
      });
    }
    var first = codec && !accepts(address) && accepts(new URL(address)) ?
      new URL(address) : address;
    return call(first).catch(function (error) {
      var message = String((error && error.message) || error || "");
      if (typeof first === "string" &&
          /Expected\s+URL\s*,\s*(?:bug\s+|but\s+)?got\s+string/i.test(message)) {
        return call(new URL(address));
      }
      if (typeof first !== "string" &&
          /Expected\s+string\s*,\s*(?:bug\s+|but\s+)?got\s+(?:object|URL)/i.test(message)) {
        return call(address);
      }
      throw error;
    });
  }

  function Client(log) {
    this.log = log || function () {};
    this.manager = null;
    this.session = null;
    this.pending = null;
    this.onEvent = null;
  }

  Client.prototype.connect = function () {
    var self = this;
    if (self.manager) { return Promise.resolve(self.manager); }
    if (self.pending) { return self.pending; }
    self.pending = loadApi().then(token).then(function (sessionToken) {
      return new Promise(function (resolve, reject) {
        var settled = false;
        var timer = setTimeout(function () { fail(new Error("AppsManager connection timed out")); }, 12000);
        function fail(error) {
          if (settled) { return; }
          settled = true;
          clearTimeout(timer);
          reject(error);
        }
        var state = {
          onsessionconnected: function () {
            global.lib_apps.AppsManager.get(self.session).then(function (manager) {
              if (settled) { return; }
              settled = true;
              clearTimeout(timer);
              self.manager = manager;
              ["APP_INSTALLING_EVENT", "APP_INSTALLED_EVENT", "APP_DOWNLOAD_FAILED_EVENT",
                "APP_UPDATING_EVENT", "APP_UPDATED_EVENT"].forEach(function (name) {
                if (manager[name] === undefined || !manager.addEventListener) { return; }
                manager.addEventListener(manager[name], function (event) {
                  if (self.onEvent) { self.onEvent(name, event); }
                });
              });
              resolve(manager);
            }, fail);
          },
          onsessiondisconnected: function () {
            self.manager = null;
            self.session = null;
            self.pending = null;
            self.log("AppsManager disconnected");
          }
        };
        try {
          self.session = new global.lib_session.Session();
          self.session.open("websocket", "localhost", sessionToken || "secrettoken", state, true);
        } catch (error) { fail(error); }
      });
    }).catch(function (error) {
      self.pending = null;
      throw error;
    });
    return self.pending;
  };

  Client.errorText = errorText;
  Client.callUrlMethod = callUrlMethod;
  global.JScalatedAppsManager = Client;
}(window));
(function () {
  "use strict";

  var APP_NAME = "Sideload";
  var UPDATE_URL = "https://sideload.s-q.top/update.webmanifest";

  var root = document.getElementById("jscalated-store-installer");
  if (root) { root.style.display = "block"; return; }

  root = document.createElement("div");
  root.id = "jscalated-store-installer";
  root.innerHTML = [
    '<style>',
    '#jscalated-store-installer{position:fixed;z-index:2147483647;inset:0;overflow:auto;background:#07111f;color:#e2e8f0;font:14px Arial,sans-serif}',
    '#jscalated-store-installer *{box-sizing:border-box}',
    '#jscalated-store-installer .wrap{max-width:580px;margin:auto;padding:15px}',
    '#jscalated-store-installer h1{color:#38bdf8;font-size:24px}',
    '#jscalated-store-installer p{line-height:1.4}',
    '#jscalated-store-installer .panel{background:#0b1728;border:1px solid #20324a;border-radius:11px;padding:13px}',
    '#jscalated-store-installer button{width:100%;margin:8px 0;padding:12px;border:0;border-radius:7px;background:#22c55e;color:#020617;font-weight:bold}',
    '#jscalated-store-installer button.secondary{background:#334155;color:#fff}',
    '#jscalated-store-installer .warning{color:#fca5a5}',
    '#jscalated-store-installer pre{min-height:95px;white-space:pre-wrap;word-break:break-word;background:#020617;color:#22c55e;padding:10px;border-radius:7px;font:11px monospace}',
    '</style>',
    '<div class="wrap"><h1 id="jsc-title">Install app</h1>',
    '<div class="panel"><p id="jsc-intro"></p>',
    '<p class="warning">The package signature is not cryptographically verified. Installation depends on this device accepting the package.</p>',
    '<button id="jsc-store-retry" type="button">Retry installation</button>',
    '<button id="jsc-store-close" class="secondary" type="button">Close</button></div>',
    '<pre id="jsc-store-log" role="status" aria-live="polite">Starting…</pre></div>'
  ].join("");
  (document.body || document.documentElement).appendChild(root);

  document.getElementById("jsc-title").textContent = "Install " + APP_NAME;
  document.getElementById("jsc-intro").textContent =
    "Install " + APP_NAME + " from " + UPDATE_URL + " through KaiOS AppsManager.";

  var logNode = document.getElementById("jsc-store-log");
  var retry = document.getElementById("jsc-store-retry");
  var busy = false;
  var client = null;

  function log(message) {
    logNode.textContent += "\n" + message;
    logNode.scrollTop = logNode.scrollHeight;
  }

  function callUrlMethod(manager, method, url, rest) {
    var address = new URL(String(url)).href;
    var args = rest || [];
    var requests = {
      installPackage: ["InstallPackageRequest", "updateUrl"],
      checkForUpdate: ["CheckForUpdateRequest", "updateUrl"],
      update: ["UpdateRequest", "manifestUrl"]
    };
    var request = requests[method];
    var codec = request && manager.wrapper_payload_message &&
      manager.wrapper_payload_message[request[0]];
    function accepts(value) {
      if (!codec || typeof codec.encode !== "function") { return false; }
      var payload = {};
      payload[request[1]] = value;
      if (method === "checkForUpdate") { payload.appsOption = args[0]; }
      try { codec.encode(payload); return true; }
      catch (_) { return false; }
    }
    function call(value) {
      return Promise.resolve().then(function () {
        return manager[method].apply(manager, [value].concat(args));
      });
    }
    var first = codec && !accepts(address) && accepts(new URL(address)) ?
      new URL(address) : address;
    return call(first).catch(function (error) {
      var message = String((error && error.message) || error || "");
      if (typeof first === "string" &&
          /Expected\s+URL\s*,\s*(?:bug\s+|but\s+)?got\s+string/i.test(message)) {
        return call(new URL(address));
      }
      if (typeof first !== "string" &&
          /Expected\s+string\s*,\s*(?:bug\s+|but\s+)?got\s+(?:object|URL)/i.test(message)) {
        return call(address);
      }
      throw error;
    });
  }

  function remoteUpdateUrl(app) {
    try { return new URL(String((app && (app.updateURL || app.updateUrl)) || "")).href; }
    catch (_) { return ""; }
  }

  function isTarget(app) {
    if (!app) { return false; }
    return remoteUpdateUrl(app) === UPDATE_URL || app.name === APP_NAME;
  }

  function start() {
    if (busy) { return; }
    busy = true;
    retry.disabled = true;
    log("Connecting to AppsManager…");
    client.connect().then(function (manager) {
      return manager.getAll().then(function (apps) {
        var existing = (apps || []).filter(isTarget)[0];
        if (existing && remoteUpdateUrl(existing) === UPDATE_URL &&
            (existing.manifestURL || existing.manifestUrl)) {
          log(APP_NAME + " is installed from this server. Checking for an update…");
          return callUrlMethod(manager, "checkForUpdate", UPDATE_URL,
            [{ autoInstall: false }]).then(function (available) {
            if (!available) {
              log(APP_NAME + " is already installed and up to date.");
              return null;
            }
            log("Updating " + APP_NAME + "…");
            return callUrlMethod(manager, "update",
              existing.manifestURL || existing.manifestUrl);
          });
        }
        if (existing) {
          log(APP_NAME + " was installed from another update URL. Trying this server; " +
            "a one-time migration may be required if the device rejects it.");
        }
        log("Installing " + APP_NAME + " from " + UPDATE_URL);
        return callUrlMethod(manager, "installPackage", UPDATE_URL);
      });
    }).then(function (result) {
      if (result) { log(APP_NAME + " installed or updated. Open it from the app list."); }
    }).catch(function (error) {
      log("Install failed: " + window.JScalatedAppsManager.errorText(error));
    }).then(function () {
      busy = false;
      retry.disabled = false;
    });
  }

  function loadClient() {
    if (!window.JScalatedAppsManager) {
      log("AppsManager helper did not load.");
      return;
    }
    client = new window.JScalatedAppsManager(log);
    client.onEvent = function (name, event) {
      if (name === "APP_DOWNLOAD_FAILED_EVENT") {
        log("Download failed: " + window.JScalatedAppsManager.errorText(event));
      } else {
        log(name.replace(/_EVENT$/, ""));
      }
    };
    start();
  }

  retry.onclick = function () { if (client) { start(); } else { loadClient(); } };
  document.getElementById("jsc-store-close").onclick = function () { root.style.display = "none"; };
  loadClient();
}());
